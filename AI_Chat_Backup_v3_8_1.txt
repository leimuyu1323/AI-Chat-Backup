停止后生成一个包含全部已保存链接和正文的大 Markdown；过程中可随时下载阶段性文件。
阶段性下载不停止任务。没有正文的链接、未完成的正文都会明确标记，避免当作完整备份。
沿用 v3.6 / v3.7 本地进度：同一浏览器、同一网站、同一账号标签可续跑，不删除旧数据。
开始时复核本轮遇到的每篇正文；旧正文仍保留，读取的新版本不会直接抹掉较长的旧版本。
面板显示链接数、正文数、当前链接、当前篇累计用时（含等待及重试）及正在读取的正文。
正文至少完成两轮上下滚动；后续整轮没有新增或变化的消息且无加载标记，才进入下一篇。
Markdown 按篇排列：标题、完整原链接、读取状态、用时、消息数及本篇问答文字放在一起，不再另列总链接清单。

“本轮读取结束”指上下边缘稳定，并非服务器全量历史或正文完整性的保证；附件原文件不包含在内。
*/

  function hash(text){let n=2166136261;for(let i=0;i<text.length;i++)n=Math.imul(n^text.charCodeAt(i),16777619);return (n>>>0).toString(16)+'-'+text.length;}
  /*
   * 把网页节点递归转换为 Markdown。递归就是函数逐层处理子节点。
   * 仅处理可读文字与部分格式；列表嵌套、引用、复杂公式可能失真，图片附件不含原文件。

  function markdown(root){
    function walk(node){if(node.nodeType===3)return node.textContent;if(node.nodeType!==1)return '';
      if(node.matches('script,style,button,svg,.sr-only,.cdk-visually-hidden,.katex-html'))return '';

      const tag=node.tagName.toLowerCase(),body=()=>[...node.childNodes].map(walk).join('');
      if(tag==='br')return '\n';
      if(tag==='pre'){const text=(node.querySelector('code')||node).textContent,ticks='`'.repeat(Math.max(3,...(text.match(/`+/g)||[]).map(s=>s.length+1)));return '\n\n'+ticks+'\n'+text+'\n'+ticks+'\n\n';}
      if(tag==='code'){const ticks='`'.repeat(Math.max(1,...(node.textContent.match(/`+/g)||[]).map(s=>s.length+1)));return ticks+' '+node.textContent+' '+ticks;}
      if(/^h[1-6]$/.test(tag))return '\n\n'+'#'.repeat(+tag[1])+' '+body()+'\n\n';
      if(tag==='a')return '['+body()+']('+(node.getAttribute('href')||'')+')';
      if(tag==='img')return '\n[图片：'+(node.alt||'原文件未下载')+']\n';
      if(['strong','b'].includes(tag))return '**'+body()+'**';
      if(['em','i'].includes(tag))return '*'+body()+'*';
      if(tag==='li')return '\n- '+body().trim()+'\n';
      if(tag==='table'){
        const rows=[...node.querySelectorAll('tr')].map(tr=>[...tr.children].map(c=>walk(c).trim().replace(/\|/g,'\\|').replace(/\n/g,' ')));
        if(!rows.length)return '';const width=Math.max(...rows.map(r=>r.length)),row=r=>'| '+Array.from({length:width},(_,i)=>r[i]||'').join(' | ')+' |';
        return '\n\n'+row(rows[0])+'\n'+row(Array(width).fill('---'))+'\n'+rows.slice(1).map(row).join('\n')+'\n\n';
      }
      return ['p','div','section','ul','ol','blockquote'].includes(tag)?'\n\n'+body()+'\n\n':body();
    }return walk(root).replace(/\n[ \t]+/g,'\n').replace(/\n{3,}/g,'\n\n').trim();
  }
  /*

  function snapshot(doc){

      const copy=node.cloneNode(true);copy.querySelectorAll('script,style,button,svg,.sr-only,.cdk-visually-hidden').forEach(n=>n.remove());
      const text=copy.textContent.trim(),md=markdown(copy);
      const owner=platform==='gemini'?(role==='user'?(node.closest('[id^="user-query-content-"]')||node.closest('user-query')||node):(node.closest('message-content')||node.closest('model-response')||node)):(node.closest('[data-message-id]')||node);
      const id=node.getAttribute('data-message-id')||owner.getAttribute('data-message-id')||owner.id||node.id;
      const fingerprint=role+':'+hash(text),occurrence=seen.get(fingerprint)||0;seen.set(fingerprint,occurrence+1);
      return {key:role+':'+(id||fingerprint+':'+occurrence),role,text,markdown:md};
    }).filter(hasText);
  }
  function onScreen(element){
    if(!element?.getClientRects().length)return false;
    const r=element.getBoundingClientRect();let left=Math.max(0,r.left),right=Math.min(window.innerWidth,r.right),top=Math.max(0,r.top),bottom=Math.min(window.innerHeight,r.bottom);
    for(let n=element;n&&n!==document.body;n=n.parentElement){
      const css=getComputedStyle(n);if(n.hidden||n.getAttribute('aria-hidden')==='true'||css.display==='none'||css.visibility==='hidden')return false;
      if(n!==element){const b=n.getBoundingClientRect();if(b.width>1&&/hidden|clip|auto|scroll/.test(css.overflowX)){left=Math.max(left,b.left);right=Math.min(right,b.right);}if(b.height>1&&/hidden|clip|auto|scroll/.test(css.overflowY)){top=Math.max(top,b.top);bottom=Math.min(bottom,b.bottom);}}
    }
    return right-left>1&&bottom-top>1;
  }
  function historyRoots(){
    const roots=[...document.querySelectorAll(platform==='gemini'?'conversations-list,#history,nav,aside,[role="navigation"]':'#history,nav,aside,[role="navigation"]')].filter(n=>!n.closest('#'+PANEL));

    return [...new Set([...roots.filter(n=>onScreen(n)||visibleLinks.some(a=>n.contains(a))),...visibleLinks])];
  }
  /*
   * 滚动容器可能随网站改版而改变：从历史链接的父级和候选子容器中寻找。
   * 选错容器会导致扫描不推进，因此不能把停滞视为全量备份成功。
   */
  function historyScroll(){
    const roots=historyRoots(),candidates=[],seen=new Set();
    const add=n=>{if(n&&!seen.has(n)){seen.add(n);candidates.push(n);}};
    // Inspect both directions: scrolling descendants and link ancestors.
    for(const root of roots){
      for(let n=root;n&&n!==document.body;n=n.parentElement)add(n);
      for(const n of root.querySelectorAll('*'))if(n.clientHeight>=30&&n.scrollHeight>n.clientHeight+5)add(n);
    }
    const usable=candidates.filter(n=>onScreen(n)&&n.clientHeight>=30&&(/auto|scroll/.test(getComputedStyle(n).overflowY)||n.scrollTop>0));
    const withLinks=usable.filter(n=>[...n.querySelectorAll('a[href]')].some(a=>urlOf(a.href)&&onScreen(a)));
    return withLinks.find(n=>n.scrollHeight>n.clientHeight+5)||withLinks[0]||usable.find(n=>n.scrollHeight>n.clientHeight+5)||usable[0]||null;
  }

  function historyFrame(sc){
    const bounds=sc.getBoundingClientRect(),rows=[],visible=[];
    for(const a of sc.querySelectorAll('a[href]')){const url=urlOf(a.href);if(!url||new URL(url).origin!==ORIGIN)continue;
      const item={url,title:a.textContent.trim()||found.get(url)?.title||'未命名对话'};rows.push(item);
      const r=a.getBoundingClientRect();if(r.height>0&&r.bottom>bounds.top&&r.top<bounds.bottom)visible.push(item);
    }
    const shown=visible.length?visible:rows.slice(-10);
    return {rows,visibleCount:visible.length,signature:shown.map(r=>r.url).join('|'),tail:shown.at(-1)?.title||'未见对话链接',top:sc.scrollTop,height:sc.scrollHeight,viewport:sc.clientHeight,
      bottom:sc.scrollTop+sc.clientHeight>=sc.scrollHeight-8,
      loading:!!sc.querySelector('[role="progressbar"],[aria-busy="true"]')||!!document.querySelector('[aria-label="Loading conversation history"],[aria-label="Loading Gems and Recent conversations"]')};
  }
  function conversationScroll(doc){
    if(platform==='gemini'){const n=doc.querySelector('infinite-scroller.chat-history');if(n)return n;}
    const first=doc.querySelector(platform==='gemini'?'user-query,model-response,.query-text':'[data-message-author-role],article[data-turn]');
    for(let n=first?.parentElement;n&&n!==doc.body;n=n.parentElement){const s=doc.defaultView.getComputedStyle(n);if(/auto|scroll/.test(s.overflowY)&&n.scrollHeight>n.clientHeight+5)return n;}
    return doc.scrollingElement;
  }
  function isLoading(doc,scroll){return !!scroll?.querySelector('[role="progressbar"],[aria-busy="true"]')||!!doc.querySelector('[data-testid="stop-button"],[aria-label="Loading conversation"],model-response[aria-busy="true"]');}
  function move(scroll,top){scroll.scrollTop=top;scroll.dispatchEvent(new scroll.ownerDocument.defaultView.Event('scroll',{bubbles:true}));}
  function merge(order,map,current,direction='append'){
    // Virtualized pages can replace every visible node between scrolls, leaving no shared anchor.
    if(order.length&&current.length&&!current.some(row=>map.has(row.key))){
      const keys=current.map(row=>row.key);if(direction==='prepend')order.unshift(...keys);else order.push(...keys);
      for(const row of current)map.set(row.key,row);return;
    }
    let next=null;
    for(let i=current.length-1;i>=0;i--){const row=current[i];if(!map.has(row.key)){const at=next===null?-1:order.indexOf(next);if(at<0)order.push(row.key);else order.splice(at,0,row.key);}map.set(row.key,row);next=row.key;}
  }

  // 保存多轮读取的差异。相同快照不重复积累；较长的旧正文不会被较短新快照覆盖丢弃。
  function combinePasses(row,prior){
    const signature=JSON.stringify(row.messages),seen=new Set([signature]),passes=[];
    for(const p of [...(prior?.previousPasses||[]),...(validRecord(prior)?[prior]:[])]){
      if(!validRecord(p))continue;const s=JSON.stringify(p.messages);if(seen.has(s))continue;seen.add(s);
      passes.push({capturedAt:p.capturedAt,messages:p.messages,loadFinished:!!p.loadFinished,stopReason:p.stopReason});
    }
    return {...row,previousPasses:passes};
  }
  async function save(row,prior){
    if(!validRecord(row))return;
    const selected=combinePasses(row,prior);
    // 写盘失败时仍能从当前页面下载已捕获内容，但不会继续处理下一篇。
    records.set(row.url,selected);showPreview(row);update();await persist(selected);return selected;
  }
  // 必须同时核对来源、账号和具体聊天 URL，不能把尚未切换完的上一篇正文记到下一篇。
  function childDoc(url){
    if(!reader||reader.closed)throw new Error('正文读取窗口已关闭或被隔离。请点击停止、下载备份，再点开始扫描重新打开。');
    let doc;try{doc=reader.document;}catch{throw new Error('无法访问正文窗口，请在该窗口恢复相同网站和原账号。');}
    if(doc.URL==='about:blank')return null;
    if(urlOf(doc.URL)!==url||doc.readyState==='loading')return null;
    guard(doc);return doc;
  }
  // 一篇可能采用懒加载或虚拟滚动：先向上读取到开头，再向下读取到结尾。
  // 合并滚动前后的消息，并定期保存部分正文；用户中途停止时也能导出。
  async function capture(item){
    const prior=records.get(item.url),map=new Map(),order=[];
    let stage='向上',signature='',stableSince=Date.now(),lastSave=0,emptySince=null,row=null;
    let round=1,previousRoundSignature=null;
    try{
      // 单篇不设总时长截止：必须上下滚动读取结束，或由用户停止。
      while(!stopped){
        guard();const doc=childDoc(item.url);
        if(!doc)throw new Error('正文窗口已离开当前链接，等待重新打开本篇。');
        const current=snapshot(doc),scroll=conversationScroll(doc);
        if(!scroll)throw new Error('当前篇的正文滚动区未识别，稍后重试本篇。');
        if(!current.length){
          emptySince??=Date.now();update('本篇正文暂未显示，正在等待，不会跳到下一篇。');
          if(Date.now()-emptySince>45000)throw new Error('当前篇暂时无法识别正文，稍后重新加载本篇。');
          await pause(POLL);continue;
        }
        emptySince=null;merge(order,map,current,stage==='向上'?'prepend':'append');
        const atEdge=stage==='向上'?scroll.scrollTop<=5:scroll.scrollTop+scroll.clientHeight>=scroll.scrollHeight-8;
        const sig=stage+'|'+current.map(m=>m.key+':'+hash(m.text)).join('|')+'|'+scroll.scrollHeight;
        if(sig!==signature||!atEdge||isLoading(doc,scroll))stableSince=Date.now();signature=sig;
        const stable=atEdge&&!isLoading(doc,scroll)&&Date.now()-stableSince>=WAIT_STABLE;
        // 一次到达底部不能直接算结束。比较两次完整上下滚动的累计消息；
        // 如果本轮发现新消息或原消息内容变化，再完整复核一轮。
        const endOfRound=stage==='向下'&&stable;
        const wholeSignature=JSON.stringify(order.map(k=>{const m=map.get(k);return [k,m.text,m.markdown];}));
        const finished=endOfRound&&previousRoundSignature!==null&&wholeSignature===previousRoundSignature;
        row={platform,accountLabel:profile,url:item.url,title:item.title||doc.title,capturedAt:new Date().toISOString(),elapsedSeconds:Math.max(0,elapsedSeconds()),verificationRounds:round,messages:order.map(k=>map.get(k)),loadFinished:finished,completeness:'unverified',stopReason:finished?'至少两轮上下滚动复核稳定；服务器端完整性未核验':'部分正文；尚未完成读取'};
        showPreview(row);phase='读取并保存正文';update('第 '+round+' 轮'+stage+'读取「'+item.title+'」：'+row.messages.length+' 条消息。');
        if(Date.now()-lastSave>=2500||finished){await save(row,prior);lastSave=Date.now();}
        if(finished)return row;
        if(endOfRound){previousRoundSignature=wholeSignature;round++;stage='向上';stableSince=Date.now();signature='';}
        else if(stage==='向上'&&stable){stage='向下';stableSince=Date.now();signature='';}
        move(scroll,stage==='向上'?Math.max(0,scroll.scrollTop-Math.max(80,scroll.clientHeight*0.7)):Math.min(scroll.scrollHeight,scroll.scrollTop+Math.max(80,scroll.clientHeight*0.7)));
        await pause(POLL);
      }
    }finally{
      if(row&&!row.loadFinished){row.elapsedSeconds=Math.max(0,elapsedSeconds());row.stopReason=stopped?'用户停止；保留已读取的部分正文':'读取遇到问题；保留部分正文并重试本篇';await save(row,prior);}
    }
    return row;
  }
  // 正文窗口一次只打开一个链接。加载超时只触发当前篇重试，不会转向下一篇。
  async function collect(item){
    guard();if(!reader||reader.closed)throw new Error('正文窗口不可用，请停止后重新点击开始扫描。');
    phase='打开当前链接';update('正在打开「'+item.title+'」，不会扫描下一篇。');
    reader.location.href=item.url;
    const since=Date.now();let doc=null;
    while(!stopped){
      guard();doc=childDoc(item.url);
      if(doc&&snapshot(doc).length)break;
      if(Date.now()-since>=PAGE_TIMEOUT)throw new Error('本篇加载超时，将继续重试本篇。');
      update('等待当前篇正文加载：'+Math.floor((Date.now()-since)/1000)+' 秒。');await pause(POLL);
    }
    if(stopped)return null;
    return capture(item);
  }

  // 当前篇的保存、读取、保存全部成功之前，外层循环不会选择下一条链接。
  async function processOne(item){
    selectItem(item);let delay=RETRY_MIN;
    while(!stopped){
      try{
        guard();phase='保存当前链接';found.set(item.url,item);update('先保存链接，再读取正文。');await persist();
        if(stopped)return;
        const row=await collect(item);
        if(stopped)return;
        if(!validRecord(row)||!row.loadFinished)throw new Error('本篇尚未读取结束，继续处理本篇。');
        failures.delete(item.url);await persist();frozenElapsed=Date.now();linkTimingActive=false;updateElapsed();sessionBodies++;update('本篇正文已保存，准备扫描下一篇。');return;
      }catch(error){
        failures.set(item.url,{...item,error:String(error.message||error),at:new Date().toISOString()});
        phase='等待重试当前篇';update(String(error.message||error)+'\n将在 '+Math.ceil(delay/1000)+' 秒后重试；未跳过本篇。');
        if(stopped)return;
        await pause(delay);delay=Math.min(RETRY_MAX,delay+RETRY_MIN);
      }
    }
  }

  // 检查当前侧栏已经渲染的候选链接；这里不保存整批，也不提前增加扫描计数。
  function visibleItems(){
    const list=[],seen=new Set();
    for(const root of historyRoots()){
      const links=root.matches('a[href]')?[root]:root.querySelectorAll('a[href]');
      for(const a of links){const url=urlOf(a.href);if(!url||new URL(url).origin!==ORIGIN||seen.has(url))continue;seen.add(url);list.push({url,title:a.textContent.trim()||found.get(url)?.title||'未命名对话'});}
    }
    return list;
  }
  function expandHistory(){
    const menu=[...document.querySelectorAll('button')].find(b=>!b.closest('#'+PANEL)&&onScreen(b)&&b.getAttribute('aria-expanded')!=='true'&&/^(Main menu|Open menu|Expand menu|Open sidebar|主菜单|打开菜单|展开菜单|打开侧边栏)$/i.test((b.getAttribute('aria-label')||'').trim()));
    menu?.click();
  }
  async function scan(){
    // seen 只在当前篇成功读完并保存后登记；失败的当前篇不会被当作处理完成。
    const seen=new Set();let nextProbe=0,probeDelay=RETRY_MIN,lastExpand=0;
    let positionedAtTop=false,bottomProbes=0;
    while(!stopped){
      try{
        guard();
        // 无论用户此前把侧栏滚到哪里，本轮都先从顶部开始，避免直接从半途向下漏掉前面的链接。
        const initialScroll=historyScroll();
        if(!positionedAtTop&&initialScroll){move(initialScroll,0);positionedAtTop=true;await pause(POLL);if(stopped)break;}
        // 优先补完已经保存链接但正文未完成的记录，再按侧栏顺序逐篇处理。
        // 只登记当前一条；不会先把整页几十个链接写进“本次扫描”计数。
        const next=[...found.values()].find(x=>!seen.has(x.url)&&!records.get(x.url)?.loadFinished)
          ||visibleItems().find(x=>!seen.has(x.url));
        if(next){
          sessionLinks++;phase='扫描当前链接';await processOne(next);
          if(stopped)break;
          seen.add(next.url);bottomProbes=0;probeDelay=RETRY_MIN;nextProbe=Date.now()+probeDelay;continue;
        }
        phase='继续扫描历史';
        const sc=historyScroll(),frame=sc?historyFrame(sc):null;
        if(!sc){
          discovery='等待侧栏显示或恢复；任务仍在运行。';update(discovery);
          if(Date.now()-lastExpand>=30000){expandHistory();lastExpand=Date.now();}
          await pause(1500);continue;
        }
        if(!frame.bottom){
          discovery='本屏链接已处理，继续向下寻找下一篇。';update(discovery);
          move(sc,Math.min(sc.scrollHeight,sc.scrollTop+Math.max(40,sc.clientHeight*0.7)));
        }else{
          discovery='到达当前已加载列表底部，持续等待和重试；不会自动结束。';update(discovery);
          if(Date.now()>=nextProbe){
            move(sc,Math.max(0,sc.scrollTop-Math.max(60,sc.clientHeight*0.6)));
            await pause(350);if(stopped)break;
            if(sc.isConnected){
              bottomProbes++;
              // 多次触底仍无新条目时，回到顶部重新逐屏核查；已完成的链接靠 seen 跳过。
              // 这样可补捉列表重排或延迟插入的链接，而不是永远只盯着底部。
              move(sc,bottomProbes%4===0?0:sc.scrollHeight);
            }
            probeDelay=Math.min(RETRY_MAX,probeDelay+RETRY_MIN);nextProbe=Date.now()+probeDelay;
          }
        }
        await pause(900);
      }catch(error){
        // 识别失败、账号异常或数据库故障时等待修复，不冒充“扫描完成”。
        phase='等待恢复';update(String(error.message||error)+'\n任务仍保持等待，可手动停止并下载。');await pause(RETRY_MIN);
      }
    }
  }

  // 每次导出都是点击当时的快照，不清空记录，不重置扫描状态，不按篇自动下载。
  function allRows(){
    const map=new Map();
    for(const r of [...imports.values(),...records.values()]){
      if(!validRecord(r))continue;const key=(r.platform||platform)+'\n'+(r.accountLabel||profile)+'\n'+r.url;
      const old=map.get(key);if(!old||textSize(r)>=textSize(old))map.set(key,r);
    }
    return [...map.values()];
  }
  // 文件链接在当前页面保留，浏览器拦截下载时可再次点击；网页无法确认磁盘写入是否成功。
  const generatedFiles=[];
  function offerFile(blob,label){
    const url=URL.createObjectURL(blob),link=add('a',undefined,files);
    const name='AI-Chat-Backup-v'+VERSION+'-'+platform+'-'+label+'-'+Date.now()+'.md';
    link.href=url;link.download=name;link.textContent=name+'（'+Math.ceil(blob.size/1024)+' KB，点击下载）';link.style.cssText='display:block;margin-top:8px;overflow-wrap:anywhere';
    generatedFiles.push({url,link});
    while(generatedFiles.length>10){const old=generatedFiles.shift();URL.revokeObjectURL(old.url);old.link.remove();}
    try{link.click();}catch{/* 若浏览器不允许自动下载，上面的文件链接仍可点击。 */}
    return name;
  }
  const inline=text=>String(text||'未命名对话').replace(/[\r\n]+/g,' ').replace(/[\\`*_{}\[\]<>#]/g,'\\$&');
  // Blob 接收分段文本，减少合成单个巨型字符串的内存开销；一个按钮只生成一个文件。
  // 每篇的原链接与对应正文放在同一节；未读链接也独立成节并明确标注状态。
  function buildMarkdown(label){
    const rows=allRows(),items=[...found.values()],chunks=[];
    const mainUrls=new Set(rows.filter(r=>(r.platform||platform)===platform&&(r.accountLabel||profile)===profile).map(r=>r.url));
    chunks.push('# AI 聊天记录备份\n\n脚本版本：v'+VERSION+'\n\n导出时间：'+new Date().toISOString()+'\n\n类型：'+label+'\n\n已保留链接：'+items.length+'；有正文的对话：'+rows.length+'。\n\n说明：仅网页已读取内容，完整性待核对；附件原文件不包含在本文件中。');
    if(!items.length&&!rows.length)chunks.push('\n\n尚未发现聊天链接。');
    for(const r of rows){
      chunks.push('\n\n---\n\n# '+inline(r.title)+'\n\n平台：'+(r.platform||platform)+'\n\n原聊天链接：<'+r.url+'>\n\n读取状态：'+(r.loadFinished?'本轮读取结束，完整性待核对':'部分正文，尚未读完')+'\n\n读取时间：'+(r.capturedAt||'未知')+'\n\n本次读取用时：'+(Number.isFinite(r.elapsedSeconds)?r.elapsedSeconds+' 秒（含等待及重试）':'旧记录未计时')+'\n\n已读取消息数：'+r.messages.filter(hasText).length+'\n\n已复核轮数：'+(r.verificationRounds||'旧记录未记录'));
      for(const m of r.messages)if(hasText(m))chunks.push('\n\n## '+(m.role==='user'?'我':m.role==='assistant'?'AI':inline(m.role))+'\n\n',m.markdown?.trim()?m.markdown:m.text);
      // 只有旧轮次包含本轮没有的消息文本时才附上，避免旧内容因新一轮读到一半而消失。
      const covered=new Set(r.messages.filter(hasText).map(m=>JSON.stringify([m.role,m.markdown||m.text])));
      for(const p of [...(r.previousPasses||[])].reverse()){
        const extra=p.messages?.filter(hasText).some(m=>!covered.has(JSON.stringify([m.role,m.markdown||m.text])));
        if(!extra)continue;
        chunks.push('\n\n### 保留的旧读取版本\n\n此版本含当前读取尚未覆盖的内容，单独保留；与上文可能重复。\n\n读取时间：'+(p.capturedAt||'未知'));
        for(const m of p.messages.filter(hasText)){chunks.push('\n\n#### '+(m.role==='user'?'我':'AI')+'\n\n',m.markdown||m.text);covered.add(JSON.stringify([m.role,m.markdown||m.text]));}
      }
    }
    const unread=items.filter(x=>!mainUrls.has(x.url));
    for(const x of unread){chunks.push('\n\n---\n\n# '+inline(x.title)+'\n\n原聊天链接：<'+x.url+'>\n\n读取状态：仅保存链接，尚未读到正文。'+(failures.has(x.url)?'\n\n当前问题：'+inline(failures.get(x.url).error):''));}
    return new Blob(chunks,{type:'text/markdown;charset=utf-8'});
  }
  function exportMarkdown(final=false){
    const blob=buildMarkdown(final?'手动停止后的汇总':'阶段性快照');
    const name=offerFile(blob,final?'stopped-all':'progress-all');
    update('已生成 '+name+'。若未自动下载，请点击下方文件链接。'+(running&&!stopped?'扫描继续运行。':''));
    return blob;
  }
  async function runSession(){
    try{await scan();}
    finally{
      // 等待当前篇的 finally 保存完毕，再生成最终文件，避免停止瞬间丢掉正在读取的正文。
      await pendingSave;
      discovery='用户手动停止；全量历史完整性未核验';
      try{await persist();}catch(error){notice='本地保存失败，当前页面仍保留内容：'+error.message;}
      if(elapsedTimer!==null){clearInterval(elapsedTimer);elapsedTimer=null;}
      if(!stopped)frozenElapsed=Date.now();running=false;updateElapsed();phase='已停止';update();
      if(!disposing){try{exportMarkdown(true);}catch(error){update('汇总文件生成失败：'+error.message+'。请保留页面，再点下载阶段性 Markdown 重试。');}}
    }
  }
