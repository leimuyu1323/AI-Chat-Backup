  /*
   * Recursively convert DOM nodes to Markdown.

  function markdown(root){
    function walk(node){if(node.nodeType===3)return node.textContent;if(node.nodeType!==1)return '';
      if(node.matches('script,style,button,svg,.sr-only,.cdk-visually-hidden,.katex-html'))return '';
      if(node.matches('.katex')){const tex=node.querySelector('annotation[encoding="application/x-tex"]');return tex?' $'+tex.textContent+'$ ':node.textContent;}
      const tag=node.tagName.toLowerCase(),body=()=>[...node.childNodes].map(walk).join('');
      if(tag==='br')return '\n';
      if(tag==='pre'){const text=(node.querySelector('code')||node).textContent,ticks='`'.repeat(Math.max(3,...(text.match(/`+/g)||[]).map(s=>s.length+1)));return '\n\n'+ticks+'\n'+text+'\n'+ticks+'\n\n';}
      if(tag==='code'){const ticks='`'.repeat(Math.max(1,...(node.textContent.match(/`+/g)||[]).map(s=>s.length+1)));return ticks+' '+node.textContent+' '+ticks;}
      if(/^h[1-6]$/.test(tag))return '\n\n'+'#'.repeat(+tag[1])+' '+body()+'\n\n';
      if(tag==='a')return '['+body()+']('+(node.getAttribute('href')||'')+')';
      if(tag==='img')return '\n[Image: '+(node.alt||'original file unavailable')+']\n';
      if(['strong','b'].includes(tag))return '**'+body()+'**';
      if(['em','i'].includes(tag))return '*'+body()+'*';
      if(tag==='li')return '\n- '+body().trim()+'\n';
      if(tag==='table'){
        const rows=[...node.querySelectorAll('tr')].map(tr=>[...tr.children].map(c=>walk(c).trim().replace(/\|/g,'\\|').replace(/\n/g,' ')));
        if(!rows.length)return '';const width=Math.max(...rows.map(r=>r.length)),row=r=>'| '+Array.from({length:width},(_,i)=>r[i]||'').join(' | ')+' |';
        return '\n\n'+row(rows[0])+'\n'+row(Array(width).fill('---'))+'\n'+rows.slice(1).map(row).join('\n')+'\n\n';
      }
      return ['p','div','section','ul','ol','blockquote'].includes(tag)?'\n\n'+body()+'\n\n':body();
    }return walk(root).replace(/\n[ \t]+/g,'\n').replace(/\n{3,}/g,'\n\n').trim();
  }

  function onScreen(element){
    if(!element?.getClientRects().length)return false;
    const r=element.getBoundingClientRect();let left=Math.max(0,r.left),right=Math.min(window.innerWidth,r.right),top=Math.max(0,r.top),bottom=Math.min(window.innerHeight,r.bottom);
    for(let n=element;n&&n!==document.body;n=n.parentElement){
      const css=getComputedStyle(n);if(n.hidden||n.getAttribute('aria-hidden')==='true'||css.display==='none'||css.visibility==='hidden')return false;
      if(n!==element){const b=n.getBoundingClientRect();if(b.width>1&&/hidden|clip|auto|scroll/.test(css.overflowX)){left=Math.max(left,b.left);right=Math.min(right,b.right);}if(b.height>1&&/hidden|clip|auto|scroll/.test(css.overflowY)){top=Math.max(top,b.top);bottom=Math.min(bottom,b.bottom);}}
    }
    return right-left>1&&bottom-top>1;
  }
  function historyRoots(){
    const roots=[...document.querySelectorAll(platform==='gemini'?'conversations-list,#history,nav,aside,[role="navigation"]':'#history,nav,aside,[role="navigation"]')].filter(n=>!n.closest('#'+PANEL));

    return [...new Set([...roots.filter(n=>onScreen(n)||visibleLinks.some(a=>n.contains(a))),...visibleLinks])];
  }
  /*
   * Find the history scroller among link ancestors and candidate containers.
   * A stalled scroll can indicate an incorrect container. It does not establish completeness.
   */
  function historyScroll(){
    const roots=historyRoots(),candidates=[],seen=new Set();
    const add=n=>{if(n&&!seen.has(n)){seen.add(n);candidates.push(n);}};
    // Inspect both directions: scrolling descendants and link ancestors.
    for(const root of roots){
      for(let n=root;n&&n!==document.body;n=n.parentElement)add(n);
      for(const n of root.querySelectorAll('*'))if(n.clientHeight>=30&&n.scrollHeight>n.clientHeight+5)add(n);
    }
    const usable=candidates.filter(n=>onScreen(n)&&n.clientHeight>=30&&(/auto|scroll/.test(getComputedStyle(n).overflowY)||n.scrollTop>0));
    const withLinks=usable.filter(n=>[...n.querySelectorAll('a[href]')].some(a=>urlOf(a.href)&&onScreen(a)));
    return withLinks.find(n=>n.scrollHeight>n.clientHeight+5)||withLinks[0]||usable.find(n=>n.scrollHeight>n.clientHeight+5)||usable[0]||null;
  }

  /*
   * Collect history links while scrolling and save progress periodically.
   * At the current bottom, retry loading older entries until the user stops the scan.
   * The scan relies on the visible page and can encounter site-side loading limits.
   */
  async function scan(){
    guard();
    const initialCount=found.size;
    const seenThisScan=new Set();
    let lastNew=Date.now(),lastSave=0,nextProbe=0,probeDelay=5000;
    scanProgress.hidden=false;
    try{
      while(!stopped){
        guard();
        const sc=historyScroll();
        const frame=sc?historyFrame(sc):null;
        if(!frame?.rows.length){
          discovery='Waiting for the sidebar. The scan is still running. Expand the Gemini history list.';
          scanProgress.textContent=discovery;
          update(discovery);
          await sleep(1500);
          continue;
        }
        const countBefore=seenThisScan.size;
        for(const item of frame.rows){
          seenThisScan.add(item.url);
          found.set(item.url,item);
        }
        if(seenThisScan.size>countBefore){
          lastNew=Date.now();
          probeDelay=5000;
          nextProbe=Date.now()+probeDelay;
        }
        const idle=Math.floor((Date.now()-lastNew)/1000);
        const position=frame.bottom?'At the bottom of the current list. Waiting and retrying.':'Scrolling to older conversations.';
        discovery=position+' History completeness is unverified.';
        scanProgress.textContent='Seen this scan: '+seenThisScan.size+' links · New: '+(found.size-initialCount)+'\n'+position+(frame.loading?' · Gemini reports loading':'')+'\nNo new links for '+idle+' seconds. Click Stop to end the scan.';
        update(discovery);
        if(Date.now()-lastSave>=5000){await persist();lastSave=Date.now();}
        if(!frame.bottom){
          // All loaded links have been collected. Jump to the bottom of Gemini's accumulated list and scan other containers screen by screen.
          const accumulated=platform==='gemini'&&sc.matches('infinite-scroller')&&sc.querySelector('conversations-list')&&frame.rows.length>Math.max(20,frame.visibleCount*2);
          move(sc,accumulated?sc.scrollHeight:Math.min(sc.scrollHeight,sc.scrollTop+Math.max(40,sc.clientHeight*0.8)));
        }else if(Date.now()>=nextProbe){
          // Move away from the bottom and back to trigger lazy loading. Back off between retries.
          move(sc,Math.max(0,sc.scrollTop-Math.max(80,sc.clientHeight*0.65)));
          await sleep(350);
          if(stopped)break;
          if(sc.isConnected)move(sc,sc.scrollHeight);
          probeDelay=Math.min(30000,probeDelay+5000);
          nextProbe=Date.now()+probeDelay;
        }
        await sleep(900);
      }
    }finally{
      discovery=stopped?'Scan stopped. Resume scanning or read the discovered links.':'The scan was interrupted. Previously saved progress is retained.';
      await persist();
      scanProgress.textContent+='\n'+discovery;
    }
    update(discovery);
  }
  /*
   * Merge snapshots from one scroll pass using shared messages as anchors.
   * With no overlap, prepend or append based on scroll direction. Virtualized lists and edits can affect ordering.
   */
  function merge(order,map,current,direction='append'){
    // Virtualized pages can replace every visible node between scrolls, leaving no shared anchor.
    if(order.length&&current.length&&!current.some(row=>map.has(row.key))){
      const keys=current.map(row=>row.key);if(direction==='prepend')order.unshift(...keys);else order.push(...keys);
      for(const row of current)map.set(row.key,row);return;
    }
    let next=null;
    for(let i=current.length-1;i>=0;i--){const row=current[i];if(!map.has(row.key)){const at=next===null?-1:order.indexOf(next);if(at<0)order.push(row.key);else order.splice(at,0,row.key);}map.set(row.key,row);next=row.key;}
  }
  /*
   * Keep differing earlier reads in previousPasses.
   * Preserve separate passes in JSON and deduplicate by comparing full message arrays.
   */
  function preservePasses(row,prior){
    const passes=[...(prior?.previousPasses||[])];
    if(validRecord(prior))passes.push({capturedAt:prior.capturedAt,messages:prior.messages,stopReason:prior.stopReason});
    const currentSignature=JSON.stringify(row.messages),seen=new Set([currentSignature]);
    return {...row,previousPasses:passes.filter(p=>{const sig=JSON.stringify(p.messages);if(seen.has(sig))return false;seen.add(sig);return true;}),completeness:'unverified'};
  }
  /*

  /*
   * Scroll up to find the start, then down to find the end. Save about every 2.5 seconds.
   * loadFinished indicates a stable scroll edge. completeness stays unverified.
   * Try to save partial content in finally after a timeout, stop, or error.
   */
  async function capture(item,read){
    const prior=records.get(item.url),map=new Map(),order=[];let row=null,stage='up',signature='',stableSince=Date.now(),lastSave=0,emptySince=null;
    const start=Date.now();
    try{while(!stopped&&Date.now()-start<CAPTURE_TIMEOUT){
      guard();const doc=read();if(!doc){await sleep(POLL);continue;}guard(doc);
      const current=snapshot(doc),scroll=conversationScroll(doc);if(!scroll)throw new Error('Conversation scroll area not detected. Open a conversation and click Check Page Detection.');
      if(!current.length){emptySince??=Date.now();if(Date.now()-emptySince>45000)throw new Error('No messages detected. Check that the reader window loaded, then use Check Page Detection.');await sleep(POLL);continue;}
      emptySince=null;merge(order,map,current,stage==='up'?'prepend':'append');
      const atEdge=stage==='up'?scroll.scrollTop<=5:scroll.scrollTop+scroll.clientHeight>=scroll.scrollHeight-8;
      const sig=stage+'|'+current.map(m=>m.key+':'+hash(m.text)).join('|')+'|'+scroll.scrollHeight;
      if(sig!==signature||!atEdge||isLoading(doc,scroll))stableSince=Date.now();signature=sig;
      const stable=atEdge&&!isLoading(doc,scroll)&&Date.now()-stableSince>=WAIT_STABLE;
      const finished=stage==='down'&&stable;
      row={platform,accountLabel:profile,url:item.url,title:item.title||doc.title,capturedAt:new Date().toISOString(),messages:order.map(k=>map.get(k)),loadFinished:finished,completeness:'unverified',stopReason:finished?'Scroll pass ended. Verify completeness.':'Partial capture. Scroll pass still in progress.'};
      if(Date.now()-lastSave>=2500||finished){await save(row,prior);lastSave=Date.now();}
      update('Scrolling '+stage+' in “'+item.title+'”: '+map.size+' messages.');
      if(finished)break;
      if(stage==='up'&&stable){stage='down';stableSince=Date.now();signature='';}
      move(scroll,stage==='up'?Math.max(0,scroll.scrollTop-Math.max(300,scroll.clientHeight*0.75)):Math.min(scroll.scrollHeight,scroll.scrollTop+Math.max(300,scroll.clientHeight*0.75)));
      await sleep(POLL);
    }}finally{if(row){if(stopped||!row.loadFinished){row.loadFinished=false;row.stopReason=stopped?'Stopped by user. Captured messages were retained.':'Timed out or encountered a page error. Partial messages were retained.';}await save(row,prior);}}

  }
